CREATE TABLE IF NOT EXISTS salarylogs(
    id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    oldsalary DECIMAL(10,2) NOT NULL,
    newsalary DECIMAL(10,2) NOT NULL,
    officrstaff_id INT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (officrstaff_id) REFERENCES officestaffs(id) ON UPDATE CASCADE ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS salarydifflogs(
    id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    diffamount DECIMAL(10,2) NOT NULL,
    officrstaff_id INT NOT NULL,
    updatedby VARCHAR(50) NOT NULL,
    FOREIGN KEY (officrstaff_id) REFERENCES officestaffs(id) ON UPDATE CASCADE ON DELETE CASCADE
)


DESC salarydifflogs;
SHOW CREATE TABLE salarydifflogs;
SHOW INDEX FROM salarydifflogs;




DELIMITER //
CREATE TRIGGER officestaffsalarydiff_bfuf
BEFORE UPDATE ON officestaffs FOR EACH ROW
FOLLOWS officestaffsalary_bfu
BEGIN

    DECLARE dbuser VARCHAR(50);

    DECLARE newsalary INT;
    DECLARE oldalary INT;
    DECLARE getdiff INT;

    SELECT USER() INTO dbuser;

    SET @newsalary := NEW.salary;
    SET @oldsalary := OLD.salary;
    SET getdiff := @newsalary - @oldsalary;


    IF OLD.salary <> NEW.salary THEN
        INSERT INTO salarydifflogs(diffamount, officrstaff_id,updatedby)
        VALUES(getdiff, OLD.id, dbuser);
    END IF;

END;

DELIMITER ;

SHOW TRIGGERS;

UPDATE officestaffs SET salary = 500000 WHERE id = 7;
UPDATE officestaffs SET salary = 300000 WHERE id = 1;

CREATE TABLE IF NOT EXISTS paydifflogs(
    id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    diffamount DECIMAL(10,2) NOT NULL,
    officrstaff_id INT NOT NULL,
    updatedby VARCHAR(50) NOT NULL,
    FOREIGN KEY (officrstaff_id) REFERENCES officestaffs(id) ON UPDATE CASCADE ON DELETE CASCADE
)


DESC paydifflogs;
SHOW CREATE TABLE paydifflogs;
SHOW INDEX FROM paydifflogs;




DELIMITER //
CREATE TRIGGER officestaffsalarydiff_bfup
BEFORE UPDATE ON officestaffs FOR EACH ROW
PRECEDES officestaffsalary_bfu
BEGIN

    SET @newsalary := NEW.salary;
    SET @oldsalary := OLD.salary;
    SET @getdiff := @newsalary - @oldsalary;


    IF OLD.salary <> NEW.salary THEN
        INSERT INTO paydifflogs(diffamount, officrstaff_id,updatedby)
        VALUES(@getdiff, OLD.id, USER());
    END IF;

END;

DELIMITER ;
