=> IN / NOT IN operators

=> IN

SELECT 10 IN (20,30,40,50,10,60,70); // 1

SELECT 10 IN (200,300,400,500,100,600,700); // 0

SELECT 'aung aung' IN ('maung maung','aung aung','zaw zaw'); // 1
SELECT 'aung aung' IN ('maung maung','Aung Aung','zaw zaw'); // 1
SELECT 'aung aung' IN ('maung maung','AUNG AUNG','zaw zaw'); // 1

SELECT 'aung' IN ('maung maung','aung aung','zaw zaw'); // 0


=> NOT IN

SELECT 10 NOT IN (20,30,40,50,10,60,70); // 1

SELECT 10 NOT IN (200,300,400,500,100,600,700); // 0

SELECT 'aung aung' NOT IN ('maung maung','aung aung','zaw zaw'); // 1
SELECT 'aung aung' NOT IN ('maung maung','Aung Aung','zaw zaw'); // 1
SELECT 'aung aung' NOT IN ('maung maung','AUNG AUNG','zaw zaw'); // 1

SELECT 'aung' NOT IN ('maung maung','aung aung','zaw zaw'); // 0

=> IN

SELECT * FROM customers;

SELECT firstname,lastname,age,city FROM customers
WHERE city = "mandalay";

SELECT firstname,lastname,age,city FROM customers
WHERE city = "mandalay" OR city = "yangon";

SELECT firstname,lastname,age,city FROM customers
WHERE city IN ("mandalay","yangon");

SELECT firstname,lastname,age,city FROM customers
WHERE city IN ("mandalay","yangon")
ORDER BY city ASC;



=> NOT IN

SELECT * FROM customers;

SELECT firstname,lastname,age,city FROM customers
WHERE city != "mandalay";

SELECT firstname,lastname,age,city FROM customers
WHERE city NOT IN "mandalay";

SELECT firstname,lastname,age,city FROM customers
WHERE city NOT IN ("mandalay","bago");

SELECT firstname,lastname,age,city FROM customers
WHERE city NOT IN ("mandalay","yangon")
ORDER BY city ASC;
