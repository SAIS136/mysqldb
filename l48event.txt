=> EVENT

=> EVENT TYPE (SCHEDULE)

(i) ONE TIME EVENT
(ii) Recurring EVENT

=> Check event ON/OFF
SHOW PROCESSLIST;

SET GLOBAL event_scheduler = ON;
SET GLOBAL event_scheduler = OFF;

---------------------------------------

=> (i) ONE TIME EVENT
->Syntax

=> Method 1

CREATE EVENT eventname
ON SCHEDULE AT TIMESTAMP
DO
statements;

=> Method 2

CREATE EVENT eventname
ON SCHEDULE AT TIMESTAMP
ON COMPLETION PRESERVE
DO
statements;



(ii) Recurring EVENT
->Syntax

=> Method 1

CREATE EVENT eventname
ON SCHEDULE EVERY INTERVAL
STARTS TIMESTAMP
DO
statements;

=> Method 2

CREATE EVENT eventname
ON SCHEDULE EVERY INTERVAL
STARTS TIMESTAMP
ENDS TIMESTAMP
DO
statements;

=> ONE TIME EVENT Method 1

CREATE EVENT eventname
ON SCHEDULE AT TIMESTAMP
DO
statements;


CREATE EVENT vipusers_ev
ON SCHEDULE AT CURRENT_TIMESTAMP + INTERVAL 30 SECOND
DO
DELETE FROM vipusers WHERE emailvarify IS NULL;

=> Lists EVENTS

SHOW EVENTS;

SELECT * FROM vipusers;

ON SCHEDULE AT TIMESTAMP
ON COMPLETION PRESERVE
DO
statements;

CREATE EVENT vipusers_ev
ON SCHEDULE AT CURRENT_TIMESTAMP + INTERVAL 30 SECOND
ON COMPLETION PRESERVE
DO

=> DROP EVENT

DROP EVENT IF EXISTS vipusers_ev;

(ii) Recurring EVENT

CREATE TABLE IF NOT EXISTS dailyifipasswords(
    id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    password VARCHAR(8) NOT NULL UNIQUE,
    status INT DEFAULT 1
);

DESC dailyifipasswords;
SHOW CREATE TABLE dailyifipasswords;
SHOW INDEX FROM dailyifipasswords;

INSERT INTO dailyifipasswords(password,status) VALUES
('1111',2),
('2222',2),
('3333',2),
('4444',2),
('5555',1);

SELECT * FROM dailyifipasswords;

UPDATE dailyifipasswords SET status = 2
WHERE status = 1;


=> Method 1

CREATE EVENT dailyifipasswords_ev
ON SCHEDULE EVERY 1 DAY
STARTS "2026-10-4 18:39:00"
DO
UPDATE dailyifipasswords SET status = 2
WHERE status = 1;

SHOW EVENTS;

SELECT * FROM dailyifipasswords;
