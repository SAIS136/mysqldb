=> UNIQUE KEY

METHOD 1 = Syntax

CREATE TABLE IF NOT EXISTS tablename(
    column1 datatype,
    column1 datatype UNIQUE,
    column1 datatype,
    ......
);

METHOD 2 = Syntax

ALTER TABLE tablename
ADD UNIQUE (column);


METHOD 3 = Syntax

=>Without keyname set
ALTER TABLE tablename
ADD  UNIQUE (column1,column2);

=>With keyname set
ALTER TABLE tablename
ADD CONSTRAINT uc_column1_column2 UNIQUE (column1,column2);

=> Without keyname set UNIQUE (METHOD 3 MULTI)

CREATE TABLE IF NOT EXISTS suppliers(
    id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    firstname VARCHAR(50) NOT NULL,
    lastname VARCHAR(50) NOT NULL,
    email VARCHAR(50) NOT NULL,
    phone VARCHAR(15) NOT NULL,
    password VARCHAR(255) NOT NULL
);

DESC suppliers;
SHOW CREATE TABLE suppliers;
SHOW INDEX FROM suppliers;

ALTER TABLE suppliers
ADD UNIQUE (email,phone);

DESC suppliers;
SHOW CREATE TABLE suppliers;
SHOW INDEX FROM suppliers;

DROP TABLE IF EXISTS suppliers;


=> With keyname set UNIQUE (METHOD 3 MULTI)

CREATE TABLE IF NOT EXISTS suppliers(
    id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    firstname VARCHAR(50) NOT NULL,
    lastname VARCHAR(50) NOT NULL,
    email VARCHAR(50) NOT NULL,
    phone VARCHAR(15) NOT NULL,
    password VARCHAR(255) NOT NULL
);

DESC suppliers;
SHOW CREATE TABLE suppliers;
SHOW INDEX FROM suppliers;

ALTER TABLE suppliers
ADD CONSTRAINT uc_email_phone UNIQUE (email,phone);

DESC suppliers;
SHOW CREATE TABLE suppliers;
SHOW INDEX FROM suppliers;
