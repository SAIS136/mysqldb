=> Insert Single Data

Method 1 = Syntax

INSERT INTO tablename (column1,column2,column3,...)
VALUE (value1,value2,value3,...);


Method 2 = Syntax

INSERT INTO tablename
VALUE (value1,value2,value3,...);



=> Insert Multi Datas

Method 1 = Syntax

INSERT INTO tablename (column1,column2,column3,...)
VALUES (value1,value2,value3,...),
        (value1,value2,value3,...),
        .............;


Method 2 = Syntax

INSERT INTO tablename
VALUES (value1,value2,value3,...),
        (value1,value2,value3,...),
        .............;
